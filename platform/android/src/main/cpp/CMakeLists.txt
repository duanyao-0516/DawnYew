cmake_minimum_required(VERSION 3.22)
project(dawnyew_crypto C)
# Optional native IDE/CMake entry point. The normal app build packages the
# four verified prebuilt JNI binaries produced by scripts/build-crypto.sh.
set(CMAKE_C_STANDARD 11)
if(NOT ANDROID)
    message(FATAL_ERROR "For host JNI use scripts/build-crypto.sh host")
endif()
if(NOT DEFINED SODIUM_PREFAB_ROOT)
    get_filename_component(SODIUM_PREFAB_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../../../../.local/crypto-work/android-prefab" ABSOLUTE)
endif()
set(SODIUM_ABI_DIR "${SODIUM_PREFAB_ROOT}/prefab/modules/sodium-static/libs/android.${ANDROID_ABI}")
add_library(sodium STATIC IMPORTED)
set_target_properties(sodium PROPERTIES IMPORTED_LOCATION "${SODIUM_ABI_DIR}/libsodium-static.a")
add_library(dawnyew_crypto SHARED dawnyew_crypto.c)
target_include_directories(dawnyew_crypto PRIVATE "${SODIUM_ABI_DIR}/include")
target_compile_options(dawnyew_crypto PRIVATE -Wall -Wextra -Werror -fvisibility=hidden)
target_link_options(dawnyew_crypto PRIVATE "-Wl,-z,max-page-size=16384" "-Wl,-z,relro,-z,now" "-Wl,--exclude-libs,ALL" "-Wl,--no-undefined")
target_link_libraries(dawnyew_crypto PRIVATE sodium)
